# Database credentials are loaded from environment variables instead of source code.
try:
    db = AnimalShelter.from_env()
    db.ensure_indexes()
    df = pd.DataFrame.from_records(db.read({}))
    total_records = db.count({})

if "_id" in df.columns:
    df.drop(columns=["_id"], inplace=True)

def build_rescue_query(filter_type):
    """Return the MongoDB query used by both the table and aggregation report."""
    if filter_type == "water":
        return {
            "animal_type": "Dog",
            "breed": {"$in": [
                "Labrador Retriever Mix",
                "Chesapeake Bay Retriever",
                "Newfoundland"
            ]},
            "sex_upon_outcome": "Intact Female",
            "age_upon_outcome_in_weeks": {"$gte": 26, "$lte": 156}
        }
    if filter_type == "mountain":
        return {
            "animal_type": "Dog",
            "breed": {"$in": [
                "German Shepherd",
                "Alaskan Malamute",
                "Old English Sheepdog",
                "Siberian Husky",
                "Rottweiler"
            ]},
            "sex_upon_outcome": "Intact Male",
            "age_upon_outcome_in_weeks": {"$gte": 26, "$lte": 156}
        }
    if filter_type == "disaster":
        return {
            "animal_type": "Dog",
            "breed": {"$in": [
                "Doberman Pinscher",
                "German Shepherd",
                "Golden Retriever",
                "Bloodhound",
                "Rottweiler"
            ]},
            "sex_upon_outcome": "Intact Male",
            "age_upon_outcome_in_weeks": {"$gte": 20, "$lte": 300}
        }
    return {}


@app.callback(
    Output("datatable-id", "data"),
    [Input("filter-type", "value")]
)
def update_dashboard(filter_type):

    query = build_rescue_query(filter_type)

    try:
        filtered_df = pd.DataFrame.from_records(db.read(query))
    except AnimalShelterError as exc:
        print(f"Unable to refresh dashboard data: {exc}")
        return []

    if "_id" in filtered_df.columns:
        filtered_df.drop(columns=["_id"], inplace=True)

    return filtered_df.to_dict("records")


@app.callback(
    Output("graph-id", "children"),
    [Input("filter-type", "value")]
)
def update_graphs(filter_type):
    # Use the same rescue filter as the table, but calculate breed counts
    # in MongoDB with an aggregation pipeline instead of pandas.
    query = build_rescue_query(filter_type)
    try:
        breed_count = pd.DataFrame.from_records(
            db.aggregate_breed_counts(query, limit=10)
        )
    except AnimalShelterError as exc:
        print(f"Unable to generate breed summary: {exc}")
        return [html.H3("Breed summary unavailable")]

    if breed_count.empty:
        return [html.H3("No breed data available")]

    return [
        dcc.Graph(
            figure=px.pie(
                breed_count,
                names="breed",
                values="count",
                title="Top 10 Breeds in Current Filter (MongoDB Aggregation)"
            )
        )
    ]


@app.callback(
    Output("datatable-id", "style_data_conditional"),
    [Input("datatable-id", "selected_columns")]
)
def update_styles(selected_columns):
    return [{
        "if": {"column_id": i},
        "background_color": "#D2F3FF"
    } for i in selected_columns]


@app.callback(
    Output("map-id", "children"),
    [
        Input("datatable-id", "derived_virtual_data"),
        Input("datatable-id", "derived_virtual_selected_rows")
    ]
)
def update_map(viewData, index):

    if viewData is None or len(viewData) == 0:
        return []

    dff = pd.DataFrame.from_dict(viewData)

    if index is None or len(index) == 0:
        row = 0
    else:
        row = index[0]

    if row >= len(dff):
        row = 0

    return [
        dl.Map(
            style={"width": "1000px", "height": "500px"},
            center=[30.75, -97.48],
            zoom=10,
            children=[
                dl.TileLayer(id="base-layer-id"),

                dl.Marker(
                    position=[
                        dff.iloc[row]["location_lat"],
                        dff.iloc[row]["location_long"]
                    ],
                    children=[
                        dl.Tooltip(dff.iloc[row]["breed"]),
                        dl.Popup([
                            html.H1("Animal Name"),
                            html.P(dff.iloc[row]["name"])
                        ])
                    ]
                )
            ]
        )
    ]


app.run_server(debug=True, port=8050)
